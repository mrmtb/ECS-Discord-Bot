{% macro section_hero(section, ctx, edit_mode) %}
{% set s = section.settings %}
{# Bottom-weighted hero padding (design.md § 5.1). The fold budget is measured
   at 1280x800: header 80 + top pad + a capped D1 + lede + one CTA row must
   leave the primary action ABOVE the fold. #}
{% set pads = {'sm': 'pt-16 pb-24 sm:pt-20 sm:pb-28',
               'md': 'pt-20 pb-28 sm:pt-24 sm:pb-36 lg:pt-28 lg:pb-40',
               'lg': 'pt-24 pb-32 sm:pt-28 sm:pb-40 lg:pt-32 lg:pb-44',
               'xl': 'pt-28 pb-40 sm:pt-32 sm:pb-48 lg:pt-36 lg:pb-56'} %}
{# The overlay IS the legibility treatment — it is the ground under white text,
   so it is ecs-green-INK, not ecs-green. There is no second bottom scrim. #}
{# The scrim is BLACK, not brand green. A green wash over a photograph does not
   read as a scrim — it desaturates the image into mud and makes every hero on
   the site the same flat rectangle (design.md § 3.3 "photo hero"). Black darkens
   without staining, so the duotone below is the ONLY thing tinting the picture
   and the two treatments stop fighting each other. #}
{% set grads = {'none': '',
                'light': 'from-black/70 via-black/35 to-transparent',
                'medium': 'from-black/75 via-black/45 to-black/10',
                'heavy': 'from-black/92 via-black/80 to-black/45'} %}
{% set img = ctx.image(s.image) if s.image else none %}
{# On screens wider than 1920 the hero's content-driven height stays ~fixed while
   the width grows, so object-cover zooms the image up and crops heads. Grow a
   min-height only ABOVE 1920 (1920 and below are unchanged) so ultrawide keeps a
   sensible aspect and the subject stays in frame. These stay until a >=2400px
   hero source lands (that asset is a user TODO, not a template fix). #}
{# A PHOTOLESS HERO'S GROUND IS THE SECTION THEME, DEFAULTING TO INK.
   This used to be a hardcoded `bg-ecs-green-ink`, which is how flat brand green
   ended up being the ground of every image-less hero, the scrim over every
   photographic one, and every closing band at the same time — the same
   rectangle four or five times per page, on nine pages. Green now buys one
   moment per page (the closing band); the openings sit on the ink ground, which
   is the page's dark anchor. A page can still ask for green explicitly with
   theme='brand'. #}
{% set hero_ground = THEME_SECTION.get(section.theme if section.theme != 'inherit' else 'dark', THEME_SECTION['dark']) %}

        {{ _duotone(s.treatment) }}
        {# Direction is RESPONSIVE. A left-to-right scrim is sized for a max-w-2xl
           text column inside a wide viewport; at 320px that column is ~90% of the
           width, so the end of every line lands on the weakest stop (1.5:1). Top-to-
           bottom puts the strong stop under the whole headline on a phone. #}
        {% if s.overlay != 'none' %}
    {% endif %}

       Why this instead of type-over-photo: an overlay hero is the single most
       common shape on the web, and it forces every headline to fight its own
       background. Splitting gives the type a clean ground to sit on (no scrim,

{% macro section_content(section, ctx, edit_mode) %}
{% set s = section.settings %}
{# slot: 'head' (design.md § 6.4) keeps a heading in the SAME 
     as the
   thing it labels, instead of stranding it in its own centred section ~96px
   and a rule away. Blocks with no slot behave exactly as they did. #}
{% set head = section.blocks | selectattr('slot', 'equalto', 'head') | list %}
{% set body = section.blocks | rejectattr('slot', 'equalto', 'head') | list %}
{# Optional per-section fill (hex-validated in the schema). #}
{%- set _fill %}{% if s.bg_color %}background-color:{{ s.bg_color }};{% endif %}{% if s.text_color %}color:{{ s.text_color }};{% endif %}{% endset -%}

    
    {# ONE LEFT EDGE DOWN THE PAGE. The outer container is max-w-7xl for EVERY
       section type — hero, content, columns, band — and the reading measure is
       an inner max-width, not a narrower centred container.

       It used to be the other way round: `mx-auto max-w-5xl` on the outer div
       put a left-aligned content section's text at x=240 on a 1440px screen
       while every hero, columns and band section started at x=112. Three
       different left edges within two screens, which is exactly what reads as
       "not aligned" even when nothing is technically broken. Centring hid it
       while everything was centred; the redesign left-aligned the page and it
       became the most visible flaw on the site.

       'wide' is now max-w-none — the outer container already IS the width. #}
    
    
        {% if s.layout == 'side-head' and head %}
        {# SIDE-HEAD. The section's title sits in a narrow left column and stays
           there while its content scrolls past; the content takes the wide right
           column and the section uses the FULL container width.

           This exists to break the band stack. Every section on the page was a
           full-width horizontal slab with its heading centred above its content,
           one after another — which is the templated page shape whether the text
           is centred or left-aligned. A two-column section with a held title
           reads as a designed page instead, and it stops a left-aligned section
           from ending 192px short of the sections either side of it.

           lg: only. Below that it is a plain stack, head then body. #}
        
    
            
    
                {{ _render_blocks(head, ctx, section, edit_mode) }}
            
    
            
    
                {{ _render_blocks(body, ctx, section, edit_mode) }}
            
    
        
    
        {% else %}
        {% if head %}
        
    
            {{ _render_blocks(head, ctx, section, edit_mode) }}
        
    
        {% endif %}
        
    
            {{ _render_blocks(body, ctx, section, edit_mode) }}
        
    
        {% endif %}
    
    

    
{% endmacro %}


{% macro section_columns(section, ctx, edit_mode) %}
{% set s = section.settings %}
{# Split Studio gutter discipline: two-column sections read as a diptych — a clear,
   generous gutter between halves and no rule. Column ORDER is deliberately left alone;
   the volunteer chose which blocks sit in which column, and silently reversing that
   would override an authoring decision they cannot undo.

   EVERY layout takes an sm: step before its md:/lg: one (C8, design.md § 11).
   The asymmetric pair go 1 -> even 2-up at sm: -> their real ratio at md:, so a
   page never changes column discipline between two adjacent sections across the
   640-767px band. (design.md § 6.1's own map omits the sm: step for these two and
   contradicts its § 11; § 11 is the binding half.)

   Every arbitrary track is minmax(0,1fr): a bare 1fr track cannot shrink below its
   image's intrinsic width, which is how a grid silently starts scrolling sideways. #}
{% set layouts = {'50-50': 'grid gap-8 sm:grid-cols-2 sm:gap-10 lg:gap-14',
                  '33-67': 'grid gap-8 sm:grid-cols-2 sm:gap-10 md:grid-cols-[minmax(0,1fr)_minmax(0,2fr)] md:gap-10 lg:gap-14',
                  '67-33': 'grid gap-8 sm:grid-cols-2 sm:gap-10 md:grid-cols-[minmax(0,2fr)_minmax(0,1fr)] md:gap-10 lg:gap-14',
                  '3col': 'grid gap-8 sm:grid-cols-2 lg:grid-cols-3 lg:gap-10'} %}
{% set ncols = 3 if s.layout == '3col' else 2 %}
{# STRETCH BY DEFAULT. Centring was the old default and it is what made the two
   division blocks fail to line up: two columns of unequal height, vertically
   centred, put their photos at different Y and the row reads as broken. Plain
   grid stretch gives columns a shared TOP edge and a shared height — which is
   also what lets a card grow and push its CTA to a common bottom baseline.
   align:'center' stays as the opt-in for a short-copy-beside-a-tall-photo row. #}
{% set valign = 'lg:items-center' if s.align == 'center' and ncols == 2 else '' %}
{% set head = section.blocks | selectattr('slot', 'equalto', 'head') | list %}
{% set body = section.blocks | rejectattr('slot', 'equalto', 'head') | list %}

    
    
    
        {% if head %}
        
    
            {{ _render_blocks(head, ctx, section, edit_mode) }}
        
    
        {% endif %}
        
    
            {% for col in range(ncols) %}
            {# Fold any block whose col exceeds the layout (e.g. a 3-col card
               after switching to a 2-col layout) into the last column, so it
               never silently vanishes from the page. #}
            {% set colblocks = [] %}
            {% for block in body if (block.col or 0) == col or (col == ncols - 1 and (block.col or 0) >= ncols) %}{% set _ = colblocks.append(block) %}{% endfor %}
            
    
                {# _render_blocks, not a bare render_block loop: RHYTHM zeroes the
                   gap between two adjacent button blocks BECAUSE a flex row is
                   supposed to be handling that spacing. Without the row, two
                   buttons in a column render stacked and touching. #}
                {{ _render_blocks(colblocks, ctx, section, edit_mode) }}
            
    
            {% endfor %}
        
    
    
    

    
{% endmacro %}
